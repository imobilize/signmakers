{{/* Home hero image carousel — modelled on the Goodwin & Goodwin hero slideshow. Slides come from data/hero.yaml (portfolio page + image). Slides are rendered by Hugo (not by Alpine) so the first image loads immediately and crawlers see real tags. */}} {{- $slides := slice -}} {{- range site.Data.hero.slides -}} {{- $cfg := . -}} {{- with site.GetPage $cfg.page -}} {{- $p := . -}} {{- with $p.Resources.GetMatch $cfg.image -}} {{- $lg := .Resize (printf "%dx webp q82" (cond (gt .Width 1600) 1600 .Width)) -}} {{- $sm := .Resize "800x webp q78" -}} {{- $slides = $slides | append (dict "lg" $lg "sm" $sm "label" $cfg.label "url" $p.RelPermalink "focus" (default "center" $cfg.focus)) -}} {{- else -}} {{- warnf "hero carousel: image %q not found in %s" $cfg.image $cfg.page -}} {{- end -}} {{- else -}} {{- warnf "hero carousel: page %q not found" $cfg.page -}} {{- end -}} {{- end -}} {{- $n := len $slides -}} {{- if gt $n 0 }}
{{- range $i, $s := $slides }}
{{ $s.label }}
{{- end }}
